function mkProj(p){
  const pr=d3[p.type]();
  if(p.rotate)pr.rotate(p.rotate);
  if(p.parallels)pr.parallels(p.parallels);
  if(p.sphere){
    const box=[[24,24],[MW-24,MH-24]];
    if(p.type==="geoMercator"){pr.fitExtent(box,{type:"MultiPoint",coordinates:[[-179.99,-66],[179.99,84]]});pr.clipExtent([pr([-179.99,84]),pr([179.99,-66])]);}
    else pr.fitExtent(box,{type:"Sphere"});
    return pr;}
  const [w,s,e,n]=p.region,c=[];
  for(let k=0;k<=20;k++)c.push([w+(e-w)*k/20,s]);
  for(let k=0;k<=20;k++)c.push([e,s+(n-s)*k/20]);
  for(let k=0;k<=20;k++)c.push([e-(e-w)*k/20,n]);
  for(let k=0;k<=20;k++)c.push([w,n-(n-s)*k/20]);
  pr.fitExtent([[0,0],[MW,MH]],{type:"LineString",coordinates:c});
  // Remplir tout le cadre : on agrandit jusqu'à ce que la région couvre l'écran dans sa plus petite dimension.
  const b=d3.geoPath(pr).bounds({type:"LineString",coordinates:c}),k=p.fit==="contain"?1:Math.min(1.35,Math.max(MW/(b[1][0]-b[0][0]),MH/(b[1][1]-b[0][1])));
  const t=pr.translate();pr.scale(pr.scale()*k).translate([MW/2+(t[0]-MW/2)*k,MH/2+(t[1]-MH/2)*k]);
  pr.clipExtent([[0,0],[MW,MH]]);
  return pr;
}

const dkm=d=>d<1?fr(Math.round(d*100)*10)+" m":d<20?fr(d,1)+" km":fr(Math.round(d/10)*10)+" km";
function topoBounds(t){
  const GW=192,rad=Math.PI/180,m0=156543.03*Math.cos(t.lat*rad),z=Math.max(1,Math.min(14,Math.round(Math.log2(m0/(t.wkm*1000/GW/4))))),n=256*2**z;
  const mx=l=>(l+180)/360*n,my=b=>{const s=Math.sin(b*rad);return (0.5-Math.log((1+s)/(1-s))/(4*Math.PI))*n};
  const ux=x=>x/n*360-180,uy=y=>Math.atan(Math.sinh(Math.PI*(1-2*y/n)))/rad;
  const wpx=t.wkm*1000/(m0/2**z),hpx=wpx*9/16,x0=mx(t.lon)-wpx/2,y0=my(t.lat)-hpx/2;
  return {z,x0,y0,wpx,hpx,w:ux(x0),e:ux(x0+wpx),n:uy(y0),s:uy(y0+hpx),mx,my};
}

  for(let i=0;i<G.length;i++){const v=(R[i]>>1)/5-100;G[i]=v;W[i]=R[i]&1;if(v<lo)lo=v;if(v>hi)hi=v}

  const eq=t.eq,ix=eq*(t.idx||5),C=d3.contours().size([GW,GH]).smooth(true).thresholds(d3.range(Math.ceil((lo+0.5)/eq)*eq,hi,eq))(G);

      h+=`<rect x="${cx-gap-x(c.m[i])}" y="${y+3}" width="${x(c.m[i])}" height="${rh-6}" fill="var(--blue)"/><rect x="${cx+gap}" y="${y+3}" width="${x(c.f[i])}" height="${rh-6}" fill="var(--accent)"/><text class="chl" x="${cx}" y="${y+rh/2+9}" text-anchor="middle">${AG[i]}</text>`}
    h+=`<text class="chs" x="${cx}" y="${MH-18}" text-anchor="middle">Part de la population totale, par groupe d'âge de cinq ans</text>`;
  }else if(c.t==="bar"){
    const L=170,R=80,T=130,B=120,pw=MW-L-R,ph=MH-T-B,mx=Math.max(...c.items.map(d=>d[1])),y0=c.y0||0,top=Math.ceil(mx/5)*5,y=v=>T+ph-(v-y0)/(top-y0)*ph,bw=pw/c.items.length;
    h+=`<text class="cht" x="${L}" y="70">${esc(c.title)}</text>`;
    for(let v=y0;v<=top;v+=5)h+=`<line class="chg" x1="${L}" x2="${MW-R}" y1="${y(v)}" y2="${y(v)}"/><text class="chl" x="${L-18}" y="${y(v)+10}" text-anchor="end">${v}</text>`;
    c.items.forEach((d,i)=>{h+=`<rect x="${L+i*bw+bw*0.2}" y="${y(d[1])}" width="${bw*0.6}" height="${y(y0)-y(d[1])}" fill="var(--blue)"/><text class="cht" x="${L+i*bw+bw/2}" y="${y(d[1])-16}" text-anchor="middle">${fr(d[1],1)}</text><text class="chl" x="${L+i*bw+bw/2}" y="${MH-B+46}" text-anchor="middle">${esc(d[0])}</text>`});
  }else{
    const R=300,cx=520,cy=480,arcs=d3.pie().sort(null).value(d=>d[1])(c.items),arc=d3.arc().innerRadius(0).outerRadius(R);
    h+=`<text class="cht" x="90" y="70">${esc(c.title)}</text><g transform="translate(${cx},${cy})">${arcs.map((a,i)=>`<path d="${arc(a)}" fill="${COLORS[i%COLORS.length]}" stroke="#fff" stroke-width="2"/>`).join("")}</g>`;
    c.items.forEach((d,i)=>{h+=`<rect x="980" y="${150+i*56}" width="38" height="38" fill="${COLORS[i%COLORS.length]}"/><text class="chl" x="1034" y="${150+i*56+29}">${esc(d[0])}</text>`});
  }
  return h+`<text class="chs" x="${MW-24}" y="${c.t==="pyr"?MH-18:MH-22}" text-anchor="end">${c.t==="pyr"?"":"Source : "+esc(c.src)}</text></svg>`;
}

// Lignes de niveau décoratives : champ lisse, tracé par carrés mobiles.
function contours(cv,seed=1){
  const dpr=Math.min(2,window.devicePixelRatio||1),W=cv.clientWidth,H=cv.clientHeight; if(!W||!H)return;
  cv.width=W*dpr;cv.height=H*dpr;const c=cv.getContext("2d");c.scale(dpr,dpr);
  const st=getComputedStyle(cv);c.strokeStyle=st.color;c.lineWidth=1;
  const f=(x,y)=>Math.sin(x*0.0041+seed)*Math.cos(y*0.0057-seed*2)+0.6*Math.sin((x+y)*0.0029+seed*3)+0.45*Math.cos(x*0.0093-y*0.0071+seed)+0.25*Math.sin(y*0.013+x*0.002);
  const g=14,nx=Math.ceil(W/g)+1,ny=Math.ceil(H/g)+1,v=[];
  for(let j=0;j<ny;j++){v[j]=[];for(let i=0;i<nx;i++)v[j][i]=f(i*g,j*g);}
  const ip=(a,b,l)=>(l-a)/(b-a);
  for(let l=-2,k=0;l<=2;l+=0.2,k++){
    c.globalAlpha=k%5===0?0.34:0.16;c.lineWidth=k%5===0?1.4:1;c.beginPath();
    for(let j=0;j<ny-1;j++)for(let i=0;i<nx-1;i++){
      const a=v[j][i],b=v[j][i+1],d=v[j+1][i],e=v[j+1][i+1],pts=[];
      if((a<l)!==(b<l))pts.push([(i+ip(a,b,l))*g,j*g]);
      if((b<l)!==(e<l))pts.push([(i+1)*g,(j+ip(b,e,l))*g]);
      if((d<l)!==(e<l))pts.push([(i+ip(d,e,l))*g,(j+1)*g]);
      if((a<l)!==(d<l))pts.push([i*g,(j+ip(a,d,l))*g]);
      for(let p=0;p+1<pts.length;p+=2){c.moveTo(pts[p][0],pts[p][1]);c.lineTo(pts[p+1][0],pts[p+1][1]);}
    }
    c.stroke();
  }
}

function connect(params,onmsg,onstatus){
  let ws,dead=false,tries=0;
  const open=()=>{
    ws=new WebSocket((location.protocol==="https:"?"wss://":"ws://")+location.host+"/ws?"+new URLSearchParams(params()));
    ws.onopen=()=>{tries=0;onstatus&&onstatus(true)};
    ws.onmessage=e=>onmsg(JSON.parse(e.data));
    ws.onclose=()=>{onstatus&&onstatus(false);if(!dead)setTimeout(open,Math.min(5000,500*++tries))};
  };
  open();
  return {send:o=>{if(ws&&ws.readyState===1)ws.send(JSON.stringify(o))},stop:()=>{dead=true;ws&&ws.close()}};
}
